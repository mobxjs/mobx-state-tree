import { types, type Instance } from "../../../../src"
import { SearchForm } from "./form"

const Forms = types.model({ search: types.optional(SearchForm, {}) })

export const SearchService = types
    .model("SearchService", { lastQuery: "", forms: types.optional(Forms, {}) })
    .actions(self => ({
        search(query: string) {
            self.lastQuery = query
        }
    }))

export interface SearchServiceType extends Instance<typeof SearchService> {}
